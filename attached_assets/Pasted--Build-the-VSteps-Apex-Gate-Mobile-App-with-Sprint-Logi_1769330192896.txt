The server.js (Nut-JS with Shift Support)This script runs on your PC. It is optimized to handle the Shift modifier alongside your standard movement keys.JavaScriptconst { keyboard, Key } = require("@nut-tree-fork/nut-js");
const express = require("express");
const app = express();
const http = require("http").Server(app);
const io = require("socket.io")(http);

// Set to 0 for instant physical keyboard feel
keyboard.config.autoDelayMs = 0;

io.on("connection", (socket) => {
    console.log("VSteps Pocket-Pro Connected");

    socket.on("input", async (data) => {
        // Handle Releases first for strict separation
        if (data.release) {
            for (let k of data.release) {
                const key = getNutKey(k);
                if (key) await keyboard.releaseKey(key);
            }
        }

        // Handle Presses
        if (data.press) {
            for (let k of data.press) {
                const key = getNutKey(k);
                if (key) await keyboard.pressKey(key);
            }
        }
    });
});

function getNutKey(k) {
    switch(k.toLowerCase()) {
        case 'w': return Key.W;
        case 'a': return Key.A;
        case 'd': return Key.D;
        case 'shift': return Key.LeftShift;
        default: return null;
    }
}

http.listen(3000, () => {
    console.log("VSteps Server running on port 3000");
});
