Part 1: The JavaScript/React Logic
This snippet demonstrates how to handle the state and filtering for the deadzone.

JavaScript
// Deadzone & Pivot Logic for VSteps
const [isStrafing, setIsStrafing] = useState(null); // 'A', 'D', or null
const DEADZONE_THRESHOLD = 15; // Degrees

const updateMovement = (currentYaw, calibratedYaw) => {
  // Normalize angle difference to -180 to 180
  let diff = currentYaw - calibratedYaw;
  if (diff > 180) diff -= 360;
  if (diff < -180) diff += 360;

  if (Math.abs(diff) > DEADZONE_THRESHOLD) {
    if (diff > 0) {
      // Pivot Right
      if (isStrafing !== 'D') {
        sendKeyToPC('D_DOWN');
        sendKeyToPC('A_UP');
        setIsStrafing('D');
      }
    } else {
      // Pivot Left
      if (isStrafing !== 'A') {
        sendKeyToPC('A_DOWN');
        sendKeyToPC('D_UP');
        setIsStrafing('A');
      }
    }
  } else {
    // Inside Deadzone - Go Straight
    if (isStrafing !== null) {
      sendKeyToPC('A_UP');
      sendKeyToPC('D_UP');
      setIsStrafing(null);
    }
  }
};
