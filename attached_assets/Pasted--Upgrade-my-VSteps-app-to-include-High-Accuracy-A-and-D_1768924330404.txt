let d_key_active = false;

const DEACTIVATE_ANGLE = 20; // Tilt needed to stop (The Hysteresis Gap)

    if (!d_key_active && gamma > ACTIVATE_ANGLE) {

        await keyboard.pressKey(Key.D);
    } 
    else if (d_key_active && gamma < DEACTIVATE_ANGLE) {
        d_key_active = false;
        await keyboard.releaseKey(Key.D);
    }

    // --- A KEY LOGIC ---
    if (!a_key_active && gamma < -ACTIVATE_ANGLE) {
        a_key_active = true;
        await keyboard.pressKey(Key.A);
    } 
    else if (a_key_active && gamma > -DEACTIVATE_ANGLE) {
        a_key_active = false;
        await keyboard.releaseKey(Key.A);
    }
});
